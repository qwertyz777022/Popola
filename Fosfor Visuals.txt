local function contrastColor(c)
	local lum = c.R * 0.299 + c.G * 0.587 + c.B * 0.114
	if lum > 0.55 then
		return Color3.fromRGB(22, 24, 28) -- тёмный текст/элемент под светлый акцент

	return WHITE

-- блик по краю: "жидкое стекло"
-- переливание по контуру: прозрачный градиент вращается внутри обводки
local function edgeShimmer(st)
	local g = new("UIGradient", {
		Rotation = 0,
		Color = ColorSequence.new(WHITE, WHITE),
		Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.05),
			NumberSequenceKeypoint.new(0.3, 0.7),
			NumberSequenceKeypoint.new(0.5, 0.95),
			NumberSequenceKeypoint.new(0.7, 0.7),
			NumberSequenceKeypoint.new(1, 0.05),
		}),
	}, st)
	local phase = math.random() * 360
	task.spawn(function()
		while st.Parent do
			for ang = 0, 359, 2 do
				if not st.Parent then
					return
				end
				g.Rotation = (ang + phase) % 360
				task.wait(0.06)
			end
		end
	end)
end

-- внутренняя рамка "жидкое стекло": тонкая, с живым переливом по контуру.
-- Для layout-контейнеров (списки/авторазмер) — обводка напрямую, чтобы не ломать layout.
local function edge(parent, thickness, transparency)
	local useRim = true
	pcall(function()
		if parent:FindFirstChildOfClass("UIListLayout") or parent:FindFirstChildOfClass("UIGridLayout") then
			useRim = false
		end
		if useRim and parent:IsA("GuiObject") and parent.AutomaticSize ~= Enum.AutomaticSize.None then
			useRim = false
		end
	end)
	local rc = 14
	pcall(function()
		local uc = parent:FindFirstChildOfClass("UICorner")
		if uc then
			rc = math.max(uc.CornerRadius.Offset - 2, 4) -- рамка повторяет скругление родителя
		end
	end)
	if not useRim then
		local st = new("UIStroke", {
			Color = WHITE,
			Thickness = 1,
			Transparency = transparency or 0.55,
			ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
		}, parent)
		edgeShimmer(st)
		return st
	end
	local rim = new("Frame", {
		Size = UDim2.new(1, -4, 1, -4),
		Position = UDim2.fromOffset(2, 2),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Active = false,
	}, parent)
	new("UICorner", {CornerRadius = UDim.new(0, rc)}, rim)
	local st = new("UIStroke", {
		Color = WHITE,
		Thickness = 1,
		Transparency = transparency or 0.55,
		ApplyStrokeMode = Enum.ApplyStrokeMode.Border,
	}, rim)
	edgeShimmer(st)
	return rim
end

local function isPress(i)
	return i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch
end

local function fadeGroup(g, on)
	if on then
		g.Visible = true
		tw(g, 0.3, {GroupTransparency = 0})
	else
		local t = tw(g, 0.3, {GroupTransparency = 1})
		t.Completed:Connect(function()
			if g.GroupTransparency >= 0.99 then
				g.Visible = false
			end
		end)
	end
end

local modules = {}

local function Module(cfg)
	local m = {
		name = cfg.name,
		ru = cfg.ru,
		key = cfg.key,
		noToggle = cfg.noToggle,
		noBind = cfg.noBind,
		statusFns = {},
		enabled = false,
		v = {},
		settings = cfg.settings or {},
		onToggle = cfg.onToggle or function() end,
		onChange = cfg.onChange or function() end,
	}
	for _, s in ipairs(m.settings) do
		m.v[s.name] = s.default
	end
	function m.changed(name, value)
		m.v[name] = value
		m.onChange(name, value)
	end
	function m.setStatus(name, str, color)
		local f = m.statusFns[name]
		if f then
			f(str, color)
		end
	end
	function m.title()
		return L(m.ru or m.name, m.name)
	end
	table.insert(modules, m)
	return m
end

local function setEnabled(mod, on)
	if mod.enabled == on then
		return
	end
	mod.enabled = on
	if mod.setVisual then
		mod.setVisual(on)
	end
	mod.onToggle(on)
	refreshKeybinds()
end

local function registerPort(def)
	local env = {
		State = nil,
		conns = {},
		insts = {},
		Colors = setmetatable({}, {__index = function() return ACCENT end}),
	}
	function env:Connect(conn)
		table.insert(self.conns, conn)
		return conn
	end
	function env:Track(inst)
		table.insert(self.insts, inst)
		return inst
	end
	function env:OnColorChange(fn)
		onAccent(function()
			pcall(fn, env)
		end)
	end
	local mod
	mod = Module({
		name = def.name,
		ru = def.ru,
		key = def.key,
		settings = def.settings or {},
		onToggle = function(on)
			if on then
				env.conns, env.insts = {}, {}
				pcall(function()
					def.onStart(env)
				end)
				if def.onRender then
					env._rc = RunService.RenderStepped:Connect(function(dt)
						if mod.enabled then
							pcall(def.onRender, env, dt)
						end
					end)
				end
				if def.onHeartbeat then
					env._bc = RunService.Heartbeat:Connect(function(dt)
						if mod.enabled then
							pcall(def.onHeartbeat, env, dt)
						end
					end)
				end
			else
				if env._rc then
					pcall(function() env._rc:Disconnect() end)
					env._rc = nil
				end
				if env._bc then
					pcall(function() env._bc:Disconnect() end)
					env._bc = nil
				end
				pcall(function()
					def.onStop(env)
				end)
				for _, c in ipairs(env.conns) do
					pcall(function() c:Disconnect() end)
				end
				for _, i in ipairs(env.insts) do
					pcall(function() i:Destroy() end)
				end
				env.conns, env.insts = {}, {}
				if env._rc then
					pcall(function() env._rc:Disconnect() end)
					env._rc = nil
				end
				if env._bc then
					pcall(function() env._bc:Disconnect() end)
					env._bc = nil
				end
			end
		end,
		onChange = function(name, value)
			if def.onChange then
				pcall(def.onChange, env, name, value)
			end
		end,
	})
	env.State = mod.v
	env.Mod = mod
	return mod
end

			local startPos = Vector2.new(i.Position.X, i.Position.Y)
			local origin = islandTarget
			islandK = 20
			startDrag(function(pos)
				islandTarget = UDim2.new(origin.X.Scale, origin.X.Offset + (pos.X - startPos.X), origin.Y.Scale, origin.Y.Offset + (pos.Y - startPos.Y))
			end, function()
				islandK = 34 -- быстрая упругая доводка после отпускания
			end)
		end
	end)
	RunService.RenderStepped:Connect(function(dt)
		local k = 1 - math.exp(-dt * islandK)
		island.Position = island.Position:Lerp(islandTarget, k)
		if islandK > 12 then
			islandK += (12 - islandK) * math.min(1, dt * 5)
		end
	end)
end

refreshKeybinds = function()
	for _, c in ipairs(kb:GetChildren()) do
		if c.Name == "Row" then
			c:Destroy()
		end
	end
	local order = 1
	for _, m in ipairs(modules) do
		if m.key and (m.enabled or not KeybindsMod.v["Only active"]) then
			local row = new("Frame", {Name = "Row", Size = UDim2.new(1, 0, 0, 18), BackgroundTransparency = 1, LayoutOrder = order}, kb)
			local col = m.enabled and kbAccent or Color3.fromRGB(190, 197, 200)
			text(row, m.title(), 12, {Size = UDim2.new(1, -40, 1, 0), TextColor3 = col, TextTruncate = Enum.TextTruncate.AtEnd})
			text(row, "[" .. m.key.Name .. "]", 12, {Size = UDim2.new(0, 40, 1, 0), Position = UDim2.new(1, -40, 0, 0), TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = col})
			order += 1
		end
	end
end
onAccent(function()
	kbAccent = ACCENT
	refreshKeybinds()
end)

-- перетаскивание Keybinds HUD
do
	local kbTarget = UDim2.new(0, 12, 0.5, 0)
	local kbK = 12
	kb.Position = kbTarget
	kb.InputBegan:Connect(function(i)
		if isPress(i) then
			local startPos = Vector2.new(i.Position.X, i.Position.Y)
			local origin = kbTarget
			kbK = 20
			startDrag(function(pos)
				kbTarget = UDim2.new(origin.X.Scale, origin.X.Offset + (pos.X - startPos.X), origin.Y.Scale, origin.Y.Offset + (pos.Y - startPos.Y))
			end, function()
				kbK = 34
			end)
		end
	end)
	RunService.RenderStepped:Connect(function(dt)
		local k = 1 - math.exp(-dt * kbK)
		kb.Position = kb.Position:Lerp(kbTarget, k)
		if kbK > 12 then
			kbK += (12 - kbK) * math.min(1, dt * 5)
		end
	end)
end

local FILM_CORE = {
	Enum.CoreGuiType.Health,
	Enum.CoreGuiType.Backpack,
	Enum.CoreGuiType.Chat,
	Enum.CoreGuiType.PlayerList,
	Enum.CoreGuiType.EmotesMenu,
}
local FilmingMod = registerPort({
	name = "Filming Mode",
	ru = "Режим съёмки",
	settings = {
		{type = "toggle", name = "Own UI too", ru = "Прятать и меню Visuals", default = false},
	},
	onStart = function(env)
		env.SavedGui = {}
		env.UiShown = false
		local function hide()
			for _, g in ipairs(playerGui:GetChildren()) do
				if g:IsA("LayerCollector") and g ~= cursorGui and g ~= toastGui then
					if g.Name == "FosforVisuals" or g.Name == "FosforVisualsDim" then
						if env.State["Own UI too"] ~= true then
							continue
						end
					end
					if env.SavedGui[g] == nil then
						env.SavedGui[g] = g.Enabled
					end
					g.Enabled = false
				end
			end
			local sg = StarterGui
			for _, t in ipairs(FILM_CORE) do
				pcall(function() sg:SetCoreGuiEnabled(t, false) end)
			end
		end
		local function show()
			for g, was in pairs(env.SavedGui or {}) do
				pcall(function()
					if g and g.Parent then
						g.Enabled = was
					end
				end)
			end
			env.SavedGui = {}
			local sg = StarterGui
			for _, t in ipairs(FILM_CORE) do
				pcall(function() sg:SetCoreGuiEnabled(t, true) end)
			end
		end
		env.Hide = hide
		env.Show = show
		hide()
		env:Connect(UIS.InputBegan:Connect(function(input, gameProcessed)
			if gameProcessed then
				return
			end
			if input.KeyCode == Enum.KeyCode.C then
				env.UiShown = not env.UiShown
				if env.UiShown then
					show()
				else
					hide()
				end
			end
		end))
		log("Режим съёмки: C — показать/скрыть UI")
	end,
	onChange = function(env, name)
		-- применяем сразу
		if name == "Own UI too" and env.Hide then
			pcall(function() env.Hide() end)
		end
	end,
	onStop = function(env)
		pcall(function() env.Show() end)
		env.UiShown = false
	end,
})

local SpecMod = registerPort({
	name = "Spectator",
	ru = "Спектатор",
	settings = {
		{type = "slider", name = "Speed", ru = "Скорость полёта", min = 10, max = 150, step = 5, default = 60},
		{type = "toggle", name = "ShiftBoost", ru = "Ускорение на Shift", default = true},
	},
	onStart = function(env)
		local cam = workspace.CurrentCamera
		if not cam then
			return
		end
		env.savedType = cam.CameraType
		env.savedSubject = cam.CameraSubject
		cam.CameraType = Enum.CameraType.Scriptable
		-- старт от текущей позиции камеры/игрока
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		env.pos = root and (root.Position + Vector3.new(0, 3, 0)) or cam.CFrame.Position
		local lv = cam.CFrame.LookVector
		env.yaw = math.atan2(-lv.X, -lv.Z)
		env.pitch = math.asin(clamp(lv.Y, -1, 1))
		env.sens = 0.004
		env:Connect(UIS.InputChanged:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseMovement then
				env.yaw = env.yaw - input.Delta.X * env.sens
				env.pitch = clamp(env.pitch - input.Delta.Y * env.sens, -1.55, 1.55)
			end
		end))
	end,
	onRender = function(env, dt)
		local cam = workspace.CurrentCamera
		if not cam then
			return
		end
		local delta = UIS:GetMouseDelta()
		if delta.X ~= 0 or delta.Y ~= 0 then
			env.yaw = env.yaw - delta.X * env.sens
			env.pitch = clamp(env.pitch - delta.Y * env.sens, -1.55, 1.55)
		end
		local cf = CFrame.fromEulerAnglesYXZ(env.pitch, env.yaw, 0)
		local move = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then
			move += cf.LookVector
		end
		if UIS:IsKeyDown(Enum.KeyCode.S) then
			move -= cf.LookVector
		end
		if UIS:IsKeyDown(Enum.KeyCode.D) then
			move += cf.RightVector
		end
		if UIS:IsKeyDown(Enum.KeyCode.A) then
			move -= cf.RightVector
		end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then
			move += Vector3.new(0, 1, 0)
		end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then
			move -= Vector3.new(0, 1, 0)
		end
		if move.Magnitude > 0 then
			local speed = tonumber(env.State.Speed) or 60
			if env.State.ShiftBoost ~= false and UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
				speed *= 2.5
			end
			env.pos += move.Unit * speed * dt
		end
		cam.CFrame = CFrame.new(env.pos) * cf
	end,
	onStop = function(env)
		pcall(function()
			local cam = workspace.CurrentCamera
			if cam then
				cam.CameraType = env.savedType or Enum.CameraType.Custom
				cam.CameraSubject = env.savedSubject
					or (player.Character and player.Character:FindFirstChildOfClass("Humanoid"))
			end
		end)
	end,
})

local EspHLMod = registerPort({
	name = "ESP Highlights",
	ru = "ESP Хайлайты",
	settings = {
		{type = "slider", name = "Fill", ru = "Прозрачность заливки %", min = 0, max = 100, step = 2, default = 68},
		{type = "slider", name = "Pulse", ru = "Скорость пульса", min = 1, max = 50, step = 1, default = 22},
	},
	onStart = function(env)
		env.Refs = {}
		local folder = Instance.new("Folder")
		folder.Name = "FosforVisualsHighlights"
		folder.Parent = workspace
		env:Track(folder)
		env.Folder = folder
		function env:build(plr)
			local old = self.Refs[plr]
			if old and old.hl then
				pcall(function() old.hl:Destroy() end)
			end
			local char = plr.Character
			local head = char and char:FindFirstChild("Head")
			if not char or not head then
				self.Refs[plr] = nil
				return
			end
			local hl = Instance.new("Highlight")
			hl.Adornee = char
			hl.FillColor = self.Colors.accent
			hl.FillTransparency = (tonumber(self.State.Fill) or 68) / 100
			hl.OutlineColor = Color3.fromRGB(255, 255, 255)
			hl.OutlineTransparency = 0
			hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
			hl.Parent = self.Folder
			self.Refs[plr] = {hl = hl, phase = math.random() * 6.28}
		end
		for _, plr in ipairs(Players:GetPlayers()) do
			env:build(plr)
			env:Connect(plr.CharacterAdded:Connect(function()
				task.wait(0.15)
				if env.enabled then
					env:build(plr)
				end
			end))
		end
		env:Connect(Players.PlayerAdded:Connect(function(plr)
			env:build(plr)
			env:Connect(plr.CharacterAdded:Connect(function()
				task.wait(0.15)
				if env.enabled then
					env:build(plr)
				end
			end))
		end))
		env:Connect(Players.PlayerRemoving:Connect(function(plr)
			local r = env.Refs[plr]
			if r and r.hl then
				pcall(function() r.hl:Destroy() end)
			end
			env.Refs[plr] = nil
		end))
		env._acc = 0
	end,
	onRender = function(env)
		local t = tick()
		local accent = env.Colors.accent
		for _, r in pairs(env.Refs) do
			if r.hl and r.hl.Parent then
				local k = (math.sin(t * ((tonumber(env.State.Pulse) or 22) / 10) + r.phase) + 1) / 2
				r.hl.FillColor = Color3.fromRGB(255, 255, 255):Lerp(accent, k)
				r.hl.FillTransparency = (tonumber(env.State.Fill) or 68) / 100
			end
		end
	end,
	onHeartbeat = function(env, dt)
		env._acc = (env._acc or 0) + dt
		if env._acc < 0.2 then
			return
		end
		env._acc = 0
		for _, plr in ipairs(Players:GetPlayers()) do
			local char = plr.Character
			local head = char and char:FindFirstChild("Head")
			if char and head then
				local r = env.Refs[plr]
				if not r or not r.hl or not r.hl.Parent or r.hl.Adornee ~= char then
					env:build(plr)
				end
			end
		end
	end,
	onChange = function(env)
		-- ничего: параметры читаются каждый кадр
	end,
	onStop = function(env)
		env.Refs = {}
	end,
})

local function weatherProps()

local function flashLoop()
	weather.flashToken += 1
	local my = weather.flashToken
	if not (WEATHERS[weather.idx].storm and weather.cc) then
		return
	end
	task.spawn(function()
		while weather.flashToken == my do
			task.wait(math.random(40, 90) / 10)
			if weather.flashToken ~= my or not weather.cc then
				break
			end
			for _ = 1, math.random(1, 2) do
				weather.cc.Brightness = 0.45
				tw(weather.cc, 0.5, {Brightness = 0}, Enum.EasingStyle.Sine)
				task.wait(0.22)
			end
		end
	end)
end

local function applyWeather(smooth)
	if not weather.cc then
		return
	end
	local w, lp, ap, cp = weatherProps()
	cancelWeatherTweens()
	rainTarget = w.rain
	if smooth then
		local info = TweenInfo.new(2.5, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)
		for _, pair in ipairs({{Lighting, lp}, {weather.atmo, ap}, {weather.cc, cp}}) do
			local t = TweenService:Create(pair[1], info, pair[2])
			t:Play()
			table.insert(weather.tweens, t)
		end
	else
		for k, v in pairs(lp) do
			Lighting[k] = v
		end
		for k, v in pairs(ap) do
			weather.atmo[k] = v
		end
		for k, v in pairs(cp) do
			weather.cc[k] = v
		end
	end
	flashLoop()
end

			elseif p then

				parts[i].CFrame = CFrame.new(0, -1e4, 0)
			end
		end
		if #list > 0 then
			workspace:BulkMoveTo(list, cfs, Enum.BulkMoveMode.FireCFrameChanged)
		end
		if n == 0 then
			folder:Destroy()
			folder, parts, pos = nil, {}, {}
		end
	end)
end

local function createTextRing(position)
	local COLOR = JumpMod.v.Color
	local DUR = 1.8
	local FADE_AT = DUR - 0.8
	local FULL = PART_SIZE * 50
	local C = FULL / 2
	local D_IN, D_OUT = FULL * 0.32, FULL * 0.52

	local part = Instance.new("Part")
	part.Name = "JumpCircleText"
	part.Anchored = true
	part.CanCollide = false
	part.CanQuery = false
	part.CanTouch = false
	part.CastShadow = false
	part.Transparency = 1
	part.Size = Vector3.new(PART_SIZE, 0.05, PART_SIZE)
	part.CFrame = CFrame.new(position + Vector3.new(0, 0.06, 0))

	local sg = Instance.new("SurfaceGui")
	sg.Face = Enum.NormalId.Top
	sg.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
	sg.PixelsPerStud = 50
	sg.LightInfluence = 0
	sg.AlwaysOnTop = false
	sg.Parent = part

	local root = new("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromOffset(FULL, FULL),
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
	}, sg)
	part.Parent = workspace

	local fadeOut = TweenInfo.new(0.7, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)

	local function ring(diameter, thickness, alpha)
		local f = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.fromOffset(diameter, diameter),
			BackgroundTransparency = 1,
			BorderSizePixel = 0,
		}, root)
		new("UICorner", {CornerRadius = UDim.new(1, 0)}, f)
		local st = new("UIStroke", {Color = COLOR, Thickness = thickness, Transparency = alpha, ApplyStrokeMode = Enum.ApplyStrokeMode.Border}, f)
		task.delay(FADE_AT, function()
			TweenService:Create(st, fadeOut, {Transparency = 1}):Play()
		end)
	end
	ring(D_IN, 2, 0.1)
	ring(D_IN, 7, 0.78)
	ring(D_OUT, 2, 0.1)
	ring(D_OUT, 7, 0.78)

	local word = "FOSFORVISUALS "
	local rMid = (D_IN + D_OUT) / 4
	local fontPx = 36
	local slots = math.floor(2 * math.pi * rMid / (fontPx * 0.95))
	local N = math.max(1, math.floor(slots / #word)) * #word
	for i = 0, N - 1 do
		local ch = word:sub(i % #word + 1, i % #word + 1)
		if ch ~= " " then
			local ang = (i / N) * 2 * math.pi
			local lbl = new("TextLabel", {
				AnchorPoint = Vector2.new(0.5, 0.5),
				Size = UDim2.fromOffset(fontPx, fontPx),
				Position = UDim2.fromOffset(C + rMid * math.sin(ang), C - rMid * math.cos(ang)),
				Rotation = math.deg(ang),
				BackgroundTransparency = 1,
				Text = ch,
				TextSize = fontPx,
				Font = Enum.Font.Michroma,
				TextColor3 = COLOR,
				TextTransparency = 0.05,
				TextStrokeColor3 = COLOR,
				TextStrokeTransparency = 0.8,
				BorderSizePixel = 0,
			}, root)
			task.delay(FADE_AT, function()
				TweenService:Create(lbl, fadeOut, {TextTransparency = 1, TextStrokeTransparency = 1}):Play()
			end)
		end
	end

	Debris:AddItem(part, DUR + 0.2)
end

local function createCircle(position)
	if JumpMod.v.Style == 2 then
		return createTextRing(position)
	end
	local COLOR = JumpMod.v.Color
	local part = Instance.new("Part")
	part.Name = "JumpCircle"
	part.Anchored = true
	part.CanCollide = false
	part.CanQuery = false
	part.CanTouch = false
	part.CastShadow = false
	part.Transparency = 1
	part.Size = Vector3.new(PART_SIZE, 0.05, PART_SIZE)
	part.CFrame = CFrame.new(position + Vector3.new(0, 0.06, 0))

	local sg = Instance.new("SurfaceGui")
	sg.Face = Enum.NormalId.Top
	sg.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
	sg.PixelsPerStud = 50
	sg.LightInfluence = 0
	sg.AlwaysOnTop = false
	sg.Parent = part

	local function makeLayer(strokeThickness)
		local frame = Instance.new("Frame")
		frame.AnchorPoint = Vector2.new(0.5, 0.5)
		frame.Position = UDim2.fromScale(0.5, 0.5)
		frame.Size = UDim2.fromScale(START_SCALE, START_SCALE)
		frame.BackgroundTransparency = 1
		frame.BorderSizePixel = 0
		frame.Parent = sg

		local c = Instance.new("UICorner")
		c.CornerRadius = UDim.new(1, 0)
		c.Parent = frame

		local stroke = Instance.new("UIStroke")
		stroke.Color = COLOR
		stroke.Thickness = strokeThickness
		stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
		stroke.Transparency = 1
		stroke.Parent = frame

		return frame, stroke
	end

	local main, mainStroke = makeLayer(4)
	local glow1, glowStroke1 = makeLayer(14)
	local glow2, glowStroke2 = makeLayer(28)

	part.Parent = workspace

	local grow = TweenInfo.new(LIFETIME, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
	local fadeIn = TweenInfo.new(FADE_IN_J, Enum.EasingStyle.Sine, Enum.EasingDirection.Out)
	local fadeOut = TweenInfo.new(LIFETIME - FADE_IN_J, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut)

	local layers = {
		{main, mainStroke, MAX_ALPHA},
		{glow1, glowStroke1, MAX_ALPHA + 0.45},
		{glow2, glowStroke2, MAX_ALPHA + 0.58},
	}

	for _, l in ipairs(layers) do
		local frame, stroke, alpha = l[1], l[2], l[3]
		TweenService:Create(frame, grow, {Size = UDim2.fromScale(END_SCALE, END_SCALE)}):Play()
		local tIn = TweenService:Create(stroke, fadeIn, {Transparency = alpha})
		tIn:Play()
		tIn.Completed:Connect(function()
			TweenService:Create(stroke, fadeOut, {Transparency = 1}):Play()
		end)
	end

	Debris:AddItem(part, LIFETIME + 0.1)
end

	end)

		pcall(FOSFOR_SHADER[p.fosfor])

	end

		{shader.atmo, p.atmo},
		{shader.cc, p.cc},
		{shader.bloom, {Intensity = p.bloom[1], Size = p.bloom[2], Threshold = p.bloom[3]}},
		{shader.rays, {Intensity = p.rays[1], Spread = p.rays[2]}},
	}

		if smooth then
			local t = TweenService:Create(pair[1], TweenInfo.new(2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), pair[2])
			t:Play()
			table.insert(shader.tweens, t)
		else
			for k, v in pairs(pair[2]) do
				pair[1][k] = v
			end
		end
	end
end

-- motion blur: размытие зависит от скорости поворота камеры
RunService.RenderStepped:Connect(function(dt)
	if not (GfxMod.enabled and gfx.motion) then
		gfx.lastLook = nil
		return
	end
	local cam = workspace.CurrentCamera
	if not cam then
		return
	end
	local look = cam.CFrame.LookVector
	local target = 0
	if GfxMod.v["Motion Blur"] and gfx.lastLook and dt > 0 then
		local ang = math.acos(math.clamp(look:Dot(gfx.lastLook), -1, 1))
		target = math.clamp(ang / dt * 2.5, 0, 12)
	end
	gfx.lastLook = look
	gfx.mblur += (target - gfx.mblur) * (1 - math.exp(-dt * 12))
	gfx.motion.Size = gfx.mblur
end)

		return
	end
	if potatoIsMine(d) then
		return
	end
	if potato.orig[d] == nil then
		potato.orig[d] = {d.Material, d.MaterialVariant}
	end
	local m = PotatoMod.v.Smooth and Enum.Material.SmoothPlastic or Enum.Material.Plastic

	end

	end
end

		end
	end
end

local function dayNightApply(env)
	if env.State.AutoReal == true then
		pcall(function()
			local t = os.date("*t")
			Lighting.ClockTime = (t.hour or 12) + (t.min or 0) / 60
		end)
	elseif env.State.EveningMorning == true then
		Lighting.ClockTime = tonumber(env.State.MorningHour) or 7
	else
		Lighting.ClockTime = tonumber(env.State.EveningHour) or 17
	end
end

local DayNightMod = registerPort({
	name = "Day / Night",
	ru = "День / Ночь",
	settings = {
		{type = "toggle", name = "EveningMorning", ru = "Вечер / Утро", default = false},
		{type = "slider", name = "NightHour", ru = "Час ночи", min = 0, max = 23, step = 1, default = 0},
		{type = "slider", name = "MorningHour", ru = "Час утра", min = 0, max = 23, step = 1, default = 7},
		{type = "slider", name = "EveningHour", ru = "Час вечера", min = 0, max = 23, step = 1, default = 17},
		{type = "toggle", name = "AutoReal", ru = "Авто по реальному времени", default = false},
	},
	onStart = function(env)
		if not env.SavedClock then
			env.SavedClock = Lighting.ClockTime
		end
		env._lastReal = nil
		dayNightApply(env)
	end,
	onChange = function(env, name)
		if name ~= nil then
			env._lastReal = nil
		end
	end,
	onHeartbeat = function(env)
		if env.State.AutoReal ~= true then
			return
		end
		local now = os.time()
		if env._lastReal and now - env._lastReal < 30 then
			return
		end
		env._lastReal = now
		dayNightApply(env)
	end,
	onStop = function(env)
		if env.SavedClock then
			pcall(function()
				Lighting.ClockTime = env.SavedClock
			end)
			env.SavedClock = nil
		end
	end,
})
	return {
		WeatherMod = WeatherMod,
		SnowMod = SnowMod,
		JumpMod = JumpMod,
		SkyMod = SkyMod,
		GraySkyMod = GraySkyMod,
		ShaderMod = ShaderMod,
		GfxMod = GfxMod,
		PotatoMod = PotatoMod,
		FullbrightMod = FullbrightMod,
		DayNightMod = DayNightMod,
	}
end)()

local skinGhost = nil

local function chamsClearParts()
	for part, st in pairs(chams.saved) do
		if part.Parent then
			part.Material = st.m
			part.Color = st.c
		end
	end
	chams.saved = {}
end

local function chamsApply()
	local char = skinGhost or player.Character
	if not char or not ChamsMod.enabled then
		return
	end
	local mode, color = ChamsMod.v.Mode, ChamsMod.v.Color
	if mode == 1 then -- ForceField
		if chams.hl then
			chams.hl:Destroy()
			chams.hl = nil
		end
		for _, d in ipairs(char:GetDescendants()) do
			if d:IsA("BasePart") and d.Name ~= "HumanoidRootPart" then
				if not chams.saved[d] then
					chams.saved[d] = {m = d.Material, c = d.Color}
				end
				if d.Material ~= Enum.Material.ForceField or d.Color ~= color then
					d.Material = Enum.Material.ForceField
					d.Color = color
				end
			end
		end
	else -- Flat / Chromatic (Highlight)
		chamsClearParts()
		if not chams.hl or chams.hl.Parent ~= char then
			if chams.hl then
				chams.hl:Destroy()
			end
			chams.hl = new("Highlight", {Name = "FosforVisualsChams", Adornee = char, DepthMode = Enum.HighlightDepthMode.Occluded}, char)
		end
		if mode == 2 then
			chams.hl.FillColor = color
			chams.hl.FillTransparency = 0
			chams.hl.OutlineTransparency = 1
		else
			chams.hl.FillTransparency = 0.2
			chams.hl.OutlineColor = WHITE
			chams.hl.OutlineTransparency = 0.2
		end
	end
end

-- PLAYER: SKINCHANGER (+ Korblox/Headless как под-тема)
--------------------------------------------------------------------
local skin = {ticket = 0, motors = {}, hide = {}, ghostParts = {}, decalOrig = {}, offset = 0, acc = 0, conn = nil, origDesc = nil}
local SkinMod
local C_OK, C_WAIT, C_ERR = Color3.fromRGB(120, 230, 150), Color3.fromRGB(255, 220, 120), Color3.fromRGB(255, 120, 120)

local function getHumanoid()
	local c = player.Character
	return c and c:FindFirstChildOfClass("Humanoid")
end

local function skinCollectReal()
	skin.motors, skin.hide = {}, {}
	local char = player.Character
	if not char or not skinGhost then
		return
	end
	local gm = {}
	for _, d in ipairs(skinGhost:GetDescendants()) do
		if d:IsA("Motor6D") and not d:FindFirstAncestorOfClass("Accessory") then
			gm[d.Name] = d
		end
	end
	for _, d in ipairs(char:GetDescendants()) do
		if d:IsA("Motor6D") and not d:FindFirstAncestorOfClass("Accessory") and not d:FindFirstAncestorOfClass("Tool") and gm[d.Name] then
			table.insert(skin.motors, {real = d, ghost = gm[d.Name]})
		elseif d:IsA("BasePart") and not d:FindFirstAncestorOfClass("Tool") then
			table.insert(skin.hide, d)
		elseif d:IsA("Decal") and not d:FindFirstAncestorOfClass("Tool") then
			if skin.decalOrig[d] == nil then
				skin.decalOrig[d] = d.Transparency
			end
			d.Transparency = 1
		end
	end
end

local function skinDestroyGhost()
	if skin.conn then
		skin.conn:Disconnect()
		skin.conn = nil
	end
	if skinGhost then
		skinGhost:Destroy()
		skinGhost = nil
	end
	for _, p in ipairs(skin.hide) do
		if p.Parent then
			p.LocalTransparencyModifier = 0
		end
	end
	for d, o in pairs(skin.decalOrig) do
		if d.Parent then
			d.Transparency = o
		end
	end
	skin.hide, skin.motors, skin.ghostParts, skin.decalOrig = {}, {}, {}, {}
end

local function skinStep(dt)
	local g, char = skinGhost, player.Character
	if not g or not char then
		return
	end
	local root = char:FindFirstChild("HumanoidRootPart")
	local groot = g.PrimaryPart
	if not root or not groot then
		return
	end
	skin.acc += dt
	if skin.acc > 0.5 then
		skin.acc = 0
		skinCollectReal()
	end
	for _, p in ipairs(skin.motors) do
		p.ghost.Transform = p.real.Transform
	end
	groot.CFrame = root.CFrame * CFrame.new(0, skin.offset, 0)
	for _, p in ipairs(skin.hide) do
		p.LocalTransparencyModifier = 1
	end
	local cam = workspace.CurrentCamera
	local close = cam and (cam.CFrame.Position - root.Position).Magnitude < 2.2
	for _, p in ipairs(skin.ghostParts) do
		p.LocalTransparencyModifier = close and 1 or 0
		p.CanCollide = false
	end
end

local function skinApply(username)
	username = (username or ""):match("^%s*(.-)%s*$")
	if username == "" then
		SkinMod.setStatus("Output", "", ACCENT)
		return
	end
	if not getHumanoid() then
		SkinMod.setStatus("Output", "No character", C_ERR)
		return
	end
	skin.ticket += 1
	local my = skin.ticket
	SkinMod.setStatus("Output", "Loading...", C_WAIT)
	task.spawn(function()
		local ok, userId = pcall(function()
			return Players:GetUserIdFromNameAsync(username)
		end)
		if not ok or not userId then
			if my == skin.ticket then
				SkinMod.setStatus("Output", "User not found", C_ERR)
			end
			return
		end
		local ok2, desc = pcall(function()
			return Players:GetHumanoidDescriptionFromUserId(userId)
		end)
		if not ok2 or not desc then
			if my == skin.ticket then
				SkinMod.setStatus("Output", "Failed to load", C_ERR)
			end
			return
		end
		if my ~= skin.ticket or not SkinMod.enabled then
			return
		end
		local hum = getHumanoid()
		if not hum then
			return
		end
		local ok3, model = pcall(function()
			return Players:CreateHumanoidModelFromDescription(desc, hum.RigType)
		end)
		if my ~= skin.ticket or not SkinMod.enabled then
			if ok3 and model then
				model:Destroy()
			end
			return
		end
		if not ok3 or not model then
			-- запасной вариант: применить описание прямо к своему персонажу (локально)
			if not skin.origDesc then
				local okd, od = pcall(function()
					return hum:GetAppliedDescription()
				end)
				if okd then
					skin.origDesc = od
				end
			end
			local ok4 = pcall(function()
				hum:ApplyDescription(desc)
			end)
			SkinMod.setStatus("Output", ok4 and "Applied" or "Blocked by game", ok4 and C_OK or C_ERR)
			return
		end

		skinDestroyGhost()
		model.Name = "FosforVisualsGhost"
		for _, d in ipairs(model:GetDescendants()) do
			if d:IsA("BasePart") then
				d.CanCollide = false
				d.CanQuery = false
				d.CanTouch = false
				d.Massless = true
				pcall(function()
					d.TopSurface = Enum.SurfaceType.Smooth
					d.BottomSurface = Enum.SurfaceType.Smooth
					d.LeftSurface = Enum.SurfaceType.Smooth
					d.RightSurface = Enum.SurfaceType.Smooth
					d.FrontSurface = Enum.SurfaceType.Smooth
					d.BackSurface = Enum.SurfaceType.Smooth
				end)
				table.insert(skin.ghostParts, d)
			end
		end
		pcall(function()
			game:GetService("ContentProvider"):PreloadAsync({model})
		end)
		if my ~= skin.ticket or not SkinMod.enabled then
			model:Destroy()
			return
		end
		local gh = model:FindFirstChildOfClass("Humanoid")
		local groot = model:FindFirstChild("HumanoidRootPart")
		if not groot then
			model:Destroy()
			SkinMod.setStatus("Output", "Failed to build", C_ERR)
			return
		end
		local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
		if root and gh then
			skin.offset = (gh.HipHeight + groot.Size.Y / 2) - (hum.HipHeight + root.Size.Y / 2)
		else
			skin.offset = 0
		end
		-- Humanoid копии расталкивает настоящего персонажа — удаляем его и скрипты
		for _, d in ipairs(model:GetDescendants()) do
			if d:IsA("Humanoid") or d:IsA("Animator") or d:IsA("BaseScript") then
				d:Destroy()
			end
		end
		groot.Anchored = true
		model.PrimaryPart = groot
		model.Parent = workspace
		skinGhost = model
		skinCollectReal()
		skin.conn = RunService.RenderStepped:Connect(skinStep)
		SkinMod.setStatus("Output", "Applied", C_OK)
	end)
end

-- ---- Korblox / Headless (под-тема Skinchanger) ----
local cosmetics = {headBackup = nil, headBackupChar = nil, respawnConn = nil}
local cosStripLeg

local function cosApplyHead(char)
	char = char or player.Character
	local head = char and char:FindFirstChild("Head")
	if not head then
		return
	end
	if not cosmetics.headBackup or cosmetics.headBackupChar ~= char then
		local bk = {decals = {}, meshes = {}, char = char}
		pcall(function()
			bk.color = head.BrickColor
			bk.material = head.Material
			bk.transparency = head.Transparency
			for _, o in ipairs(head:GetChildren()) do
				if o:IsA("Decal") then
					table.insert(bk.decals, {tex = o.Texture, face = o.Face})
				elseif o:IsA("SpecialMesh") then
					table.insert(bk.meshes, {mtype = o.MeshType, id = o.MeshId, tex = o.TextureId, scale = o.Scale, off = o.Offset, name = o.Name})
				end
			end
		end)
		cosmetics.headBackup = bk
		cosmetics.headBackupChar = char
	end
	for _, o in ipairs(head:GetChildren()) do
		if o:IsA("Decal") or o:IsA("SpecialMesh") then
			o:Destroy()
		end
	end
	local m = Instance.new("SpecialMesh")
	m.Name = "FosforHeadlessMesh"
	m.MeshType = Enum.MeshType.FileMesh
	m.MeshId = "rbxassetid://134082579"
	m.TextureId = "rbxassetid://134082627"
	m.Scale = Vector3.new(1.25, 1.25, 1.25)
	m.Parent = head
	head.BrickColor = BrickColor.new("Really black")
	head.Material = Enum.Material.Plastic
	head.Transparency = SkinMod.v.InvisibleHead and 1 or 0.1
end

local function cosStripHead(char)
	char = char or player.Character
	local head = char and char:FindFirstChild("Head")
	if not head then
		return
	end
	local pm = head:FindFirstChild("FosforHeadlessMesh")
	if pm then
		pm:Destroy()
	end
	local bk = (cosmetics.headBackup and cosmetics.headBackupChar == char) and cosmetics.headBackup or nil
	pcall(function()
		if bk then
			head.BrickColor = bk.color
			head.Material = bk.material
			head.Transparency = bk.transparency
			for _, d in ipairs(bk.decals) do
				local nd = Instance.new("Decal")
				nd.Texture = d.tex
				nd.Face = d.face
				nd.Parent = head
			end
			for _, md in ipairs(bk.meshes) do
				local nm = Instance.new("SpecialMesh")
				nm.Name = md.name
				nm.MeshType = md.mtype
				nm.MeshId = md.id
				nm.TextureId = md.tex
				nm.Scale = md.scale
				nm.Offset = md.off
				nm.Parent = head
			end
		else
			head.Transparency = 0
		end
	end)
	if bk then
		cosmetics.headBackup = nil
		cosmetics.headBackupChar = nil
	end
end

local function petLoad(def)
	local model = Instance.new("Model")
	model.Name = def.key
	local loaded = false
	local ok, res = pcall(function()
		return InsertService:LoadAsset(def.id)
	end)
	if ok and res then
		for _, c in ipairs(res:GetChildren()) do
			c.Parent = model
		end
		res:Destroy()
		loaded = true
	else
		local ok2, objs = pcall(function()
			return game:GetObjects("rbxassetid://" .. def.id)
		end)
		if ok2 and objs then
			for _, o in ipairs(objs) do
				o.Parent = model
			end
			loaded = true
		end
	end
	local partsCount = 0
	if loaded then
		for _, d in ipairs(model:GetDescendants()) do
			if d:IsA("BaseScript") or d:IsA("Humanoid") or d:IsA("Animator") or d:IsA("Sound") then
				d:Destroy()
			elseif d:IsA("BasePart") then
				d.Anchored = true
				d.CanCollide = false
				d.CanQuery = false
				d.CanTouch = false
				d.Massless = true
				partsCount += 1
			end
		end
	end
	if partsCount == 0 then
		model:Destroy()
		model = petFallback(def.key)
	end
	return model
end

local function petRescale(pet)
	local target = pet.def.height * PetsMod.v.Size
	pcall(function()
		pet.model:ScaleTo(target / pet.baseH)
	end)
	local cf, sz = pet.model:GetBoundingBox()
	pet.offset = pet.model:GetPivot():ToObjectSpace(cf)
	pet.halfH = sz.Y / 2
end

local function petSpawn(def)
	if pets.list[def.key] or pets.loading[def.key] then
		return
	end
	pets.loading[def.key] = true
	task.spawn(function()
		local model = petLoad(def)
		pets.loading[def.key] = nil
		if not PetsMod.enabled or not PetsMod.v[def.key] or pets.list[def.key] or not pets.folder then
			model:Destroy()
			return
		end
		local pet = {model = model, def = def, pos = nil, y = nil, dir = Vector3.new(0, 0, -1)}
		local okS, sc = pcall(function()
			return model:GetScale()
		end)
		local _, sz = model:GetBoundingBox()
		pet.baseH = math.max(sz.Y / ((okS and sc) or 1), 0.1)
		petRescale(pet)
		model.Parent = pets.folder
		pets.list[def.key] = pet
	end)
end

local function petSync()
	if not PetsMod.enabled then
		return
	end
	for _, def in ipairs(PET_DEFS) do
		if PetsMod.v[def.key] then
			petSpawn(def)
		else
			local pet = pets.list[def.key]
			if pet then
				pet.model:Destroy()
				pets.list[def.key] = nil
			end
		end
	end
end

local function petStep(dt)
	local char = player.Character
	local root = char and char:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end
	pets.t += dt
	local vel = root.AssemblyLinearVelocity
	local flatVel = Vector3.new(vel.X, 0, vel.Z)
	local moving = flatVel.Magnitude > 2
	pets.move += ((moving and 1 or 0) - pets.move) * (1 - math.exp(-dt * 3))
	local lv, rv = root.CFrame.LookVector, root.CFrame.RightVector
	local look = Vector3.new(lv.X, 0, lv.Z)
	look = look.Magnitude > 0.01 and look.Unit or Vector3.new(0, 0, -1)
	local right = Vector3.new(rv.X, 0, rv.Z)
	right = right.Magnitude > 0.01 and right.Unit or Vector3.new(1, 0, 0)
	if moving then
		pets.dir = pets.dir:Lerp(flatVel.Unit, 1 - math.exp(-dt * 6))
	end
	local mdir = pets.dir.Magnitude > 0.01 and pets.dir.Unit or look
	petRay.FilterDescendantsInstances = {char, pets.folder}
	local rp = root.Position
	local base = Vector3.new(rp.X, 0, rp.Z)
	local idx = 0
	for _, def in ipairs(PET_DEFS) do
		local pet = pets.list[def.key]
		if pet then
			idx += 1
			local idle = right * (3 + (idx - 1) * 2.4) - look * 0.6
			local mv = -mdir * (3.6 + (idx - 1) * 2.6) + right * ((idx % 2 == 0) and 1.3 or -1.3)
			local target = base + idle:Lerp(mv, pets.move)
			if not pet.pos or (pet.pos - base).Magnitude > 70 then
				pet.pos = target
			end
			local old = pet.pos
			pet.pos = old + (target - old) * (1 - math.exp(-dt * (2.2 + pets.move)))
			local delta = pet.pos - old
			local walking = delta.Magnitude / math.max(dt, 1e-3) > 0.8
			local want = walking and delta.Unit or look
			pet.dir = pet.dir:Lerp(want, 1 - math.exp(-dt * 8))
			if pet.dir.Magnitude < 0.01 then
				pet.dir = look
			end
			local hit = workspace:Raycast(Vector3.new(pet.pos.X, rp.Y + 3, pet.pos.Z), Vector3.new(0, -40, 0), petRay)
			local gy = hit and hit.Position.Y or (rp.Y - 3)
			pet.y = pet.y and (pet.y + (gy - pet.y) * (1 - math.exp(-dt * 12))) or gy
			local hop = walking and math.abs(math.sin(pets.t * 9 + idx)) * 0.22 or 0
			local face = CFrame.lookAt(Vector3.zero, pet.dir.Unit) * CFrame.Angles(0, math.rad(PetsMod.v.Rotation), 0)
			local bb = CFrame.new(pet.pos.X, pet.y + pet.halfH + hop, pet.pos.Z) * face
			pet.model:PivotTo(bb * pet.offset:Inverse())
		end
	end
end

					return true
				end

				for _ = 1, 6 do
					node = node.Parent

						return false

					if node == ch then
						return true

	reg({

			local folder = Instance.new("Folder")
			folder.Name = "FosforVisualsItemHL"
			folder.Parent = workspace
			env:Track(folder)
			env.Folder = folder
			local function countHl()
				local n = 0
				for _ in pairs(env.HLs) do
					n = n + 1
				end
				return n
			end
			local function addHighlight(root)
				if not root or env.HLs[root] then
					return
				end
				if countHl() >= 35 then
					return
				end
				local ok, hl = pcall(function()
					local h = Instance.new("Highlight")
					h.FillColor = Color3.fromRGB(255, 255, 255)
					h.FillTransparency = (tonumber(env.State.Fill) or 85) / 100
					h.OutlineColor = Color3.fromRGB(255, 255, 255)
					h.OutlineTransparency = (tonumber(env.State.Outline) or 35) / 100

					env.HLs[root] = {hl = hl, phase = math.random() * 6.28}
				end

				pcall(function()
					local chars = Lib.charsSet()

						return

					for _, d in ipairs(workspace:GetDescendants()) do
						if d and d.Parent and d:IsA("BasePart") and not d.Anchored and d.Transparency < 1 then

											roots[root] = true
											table.insert(order, root)
											if #order >= 60 then
												break
											end
										end
									end
								end
							end
						end
					end
					local valid = {}
					for _, root in ipairs(order) do
						if root.Parent then

									addHighlight(root)
								end
							end
						end
					end
					for root, data in pairs(env.HLs) do
						if not valid[root] or not root.Parent then
							pcall(function() data.hl:Destroy() end)
							env.HLs[root] = nil
						end
					end
					env._dirty = false
					env._lastRescan = tick()

						end
					end)

							task.wait(0.2)
							v = d.Value
						end
						if type(v) ~= "string" or v == "" then
							return
						end
						if not env.Mod.enabled then
							return
						end
						if env.State.OnlyMine == true and v ~= player.Name and v ~= player.DisplayName then
							return
						end
						if not Players:FindFirstChild(v) then
							return
						end
						if not d.Parent then
							return
						end
						local chars = Lib.charsSet()
						if Lib.inAnyCharacter(d, chars) then
							return
						end
						local root = Lib.cheapRootOf(d, chars)
						if root then
							addHighlight(root)
						end
					end)
				end
			end))
			env:Connect(workspace.DescendantRemoving:Connect(function(d)
				if not env.Mod.enabled then
					return
				end
				local hl = env.HLs and env.HLs[d]
				if hl then
					pcall(function() hl.hl:Destroy() end)
					env.HLs[d] = nil
				end
			end))
			env._acc = 0
		end,

				end
			end

			env.HLs = {}
			env.Seen = nil
			env._dirty = nil

			spinner.Position = UDim2.new(0.5, 0, 0.5, 0)

			spinner.BackgroundTransparency = 1
			spinner.Parent = bg
			local scaleU = Instance.new("UIScale")
			scaleU.Scale = 0.8
			scaleU.Parent = spinner
			local pieces = {}
			local CORNER, THICK = 52, 12
			local function addBar(cf, bSize, bPos)
				local b = Instance.new("Frame")
				b.Size = bSize
				b.Position = bPos
				b.BackgroundColor3 = env.Colors.accent
				b.BorderSizePixel = 0
				b.Parent = cf
				corner(b, 5)

				s.Color = env.Colors.accent
				s.Thickness = 1.5
				s.Transparency = 0.45
				s.Parent = b
				table.insert(pieces, {bar = b, stroke = s})
			end
			local function mk(kind)
				local c = Instance.new("Frame")
				c.Size = UDim2.fromOffset(CORNER, CORNER)
				c.BackgroundTransparency = 1
				local isTop = (kind == "TL" or kind == "TR")
				local isLeft = (kind == "TL" or kind == "BL")
				if kind == "TL" then
					c.Position = UDim2.new(0, 0, 0, 0)
				elseif kind == "TR" then
					c.Position = UDim2.new(1, -CORNER, 0, 0)
				elseif kind == "BL" then
					c.Position = UDim2.new(0, 0, 1, -CORNER)
				else
					c.Position = UDim2.new(1, -CORNER, 1, -CORNER)
				end
				c.Parent = spinner
				addBar(c, UDim2.fromOffset(CORNER, THICK), isTop and UDim2.new(0, 0, 0, 0) or UDim2.new(0, 0, 1, -THICK))
				addBar(c, UDim2.fromOffset(THICK, CORNER), isLeft and UDim2.new(0, 0, 0, 0) or UDim2.new(1, -THICK, 0, 0))
			end

			mk("TR")
			mk("BL")
			mk("BR")

			function env:FindHeld()
				local chars = Lib.charsSet()
				local c = workspace.CurrentCamera
				if not c then
					return nil
				end
				local held = nil
				pcall(function()
					for _, child in ipairs(workspace:GetChildren()) do
						if child.Name == "GrabParts" then
							local drag = child:FindFirstChild("DragPart")
							if drag and drag:IsA("BasePart") and (drag.Position - c.CFrame.Position).Magnitude < 32 then
								local gp = child:FindFirstChild("GrabPart")
								local weld = gp and gp:FindFirstChildOfClass("WeldConstraint")
								if weld then
									local candidates = {weld.Part1, weld.Part0}
									for _, part in ipairs(candidates) do
										if part and part ~= drag and not Lib.inAnyCharacter(part, chars) then
											held = Lib.cheapRootOf(part, chars) or part
											break
										end
									end
								end
								if held then
									break
								end
							end
						end
					end
				end)
				return held
			end
		end,
		onRender = function(env, dt)
			local f = env.Frame
			if not f then
				return
			end
			dt = dt or 0.016
			local held = env:FindHeld()
			local targetAlpha = 0
			if held and held.Parent then
				targetAlpha = 1
				if env.Held ~= held then
					env.Held = held
					f.gui.Adornee = held
				end
			end
			local dur = (targetAlpha > f.alpha) and env.FADE_IN or env.FADE_OUT
			dur = math.max(0.01, dur)
			local step = dt / dur
			local diff = targetAlpha - f.alpha
			if math.abs(diff) <= step then
				f.alpha = targetAlpha
			else
				f.alpha = f.alpha + math.sign(diff) * step
			end
			if f.alpha <= 0.01 then
				f.gui.Enabled = false
				env.Held = nil
				return
			end
			f.gui.Enabled = true
			f.rot = f.rot + (tonumber(env.State.RotSpeed) or 30) * dt
			f.spinner.Rotation = f.rot
			local t = tick()
			local k = ((math.sin(t * 1.6) + 1) / 2) ^ 3
			local col = env.Colors.accent:Lerp(Color3.fromRGB(255, 255, 255), k * 0.9)
			local a = f.alpha
			for _, piece in ipairs(f.pieces) do
				piece.bar.BackgroundColor3 = col
				piece.bar.BackgroundTransparency = 1 - a
				piece.stroke.Color = col
				piece.stroke.Transparency = 1 - a * 0.55
			end
		end,
		onStop = function(env)
			env.Frame = nil
			env.Held = nil
		end,
	})

	local function boardIsOwned(object)
		if object:GetAttribute("Owner") == player.Name then
			return true
		end
		local ow = object:FindFirstChild("Owner")
		if ow and ow.Value == player then
			return true
		end
		local char = player.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		if hrp then
			local pp = object:IsA("BasePart") and object or object:FindFirstChildWhichIsA("BasePart")
			if pp and (pp.Position - hrp.Position).Magnitude <= 20 then
				return true
			end
		end
		return false
	end
	local function boardApplyTo(mod, object)
		local env = mod.env
		local boardColor = env.State.Board
		local textColor = env.State.TextColor
		local label = tostring(env.State.Label or "")
		local function paint(part)
			if not part:IsA("BasePart") then
				return
			end
			for _, child in ipairs(part:GetChildren()) do
				if child:IsA("Decal") or child:IsA("Texture") or child.Name == "FosforVisualsBoardText" then
					child:Destroy()
				end
			end
			pcall(function()
				TweenService:Create(part, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {Color = boardColor}):Play()
			end)
		end
		if object:IsA("BasePart") then
			paint(object)
		end
		for _, child in ipairs(object:GetDescendants()) do
			paint(child)
		end
		local centerPart = object:IsA("BasePart") and object or nil
		local maxArea = 0
		for _, child in ipairs(object:GetDescendants()) do
			if child:IsA("BasePart") then
				local area = child.Size.X * child.Size.Z
				if area > maxArea then
					maxArea = area
					centerPart = child
				end
			end
		end
		if label ~= "" and centerPart then
			local sg = Instance.new("SurfaceGui")
			sg.Name = "FosforVisualsBoardText"
			sg.Face = Enum.NormalId.Top
			sg.SizingMode = Enum.SurfaceGuiSizingMode.PixelsPerStud
			sg.PixelsPerStud = 50
			sg.Parent = centerPart
			local lbl = Instance.new("TextLabel")
			lbl.AnchorPoint = Vector2.new(0.5, 0.5)
			lbl.Position = UDim2.fromScale(0.5, 0.5)
			lbl.Size = UDim2.new(0.9, 0, 0.8, 0)
			lbl.BackgroundTransparency = 1
			lbl.Text = label
			lbl.TextColor3 = textColor
			lbl.TextTransparency = 1
			lbl.TextScaled = true
			lbl.Font = Enum.Font.GothamBold
			lbl.Parent = sg
			TweenService:Create(lbl, TweenInfo.new(1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {TextTransparency = 0}):Play()
		end
	end

local PanicMod = Module({
	name = "Panic",
	ru = "Паника",
	noToggle = true,
	settings = {
		{type = "toggle", name = "RestoreLight", ru = "Возвращать свет", default = true},
	},
})
function PanicMod:PanicAction()
	local hidden = not self.Hidden
	self.Hidden = hidden
	if hidden then
		for _, m in ipairs(modules) do
			if m ~= self and m.enabled then
				pcall(function()
					setEnabled(m, false)
				end)
			end
		end
		if self.v.RestoreLight ~= false then
			pcall(function()
				Lighting.ClockTime = bootLighting.ClockTime
				Lighting.Ambient = bootLighting.Ambient
				Lighting.OutdoorAmbient = bootLighting.OutdoorAmbient
			end)
		end
		self.SavedGui = {}
		for _, g in ipairs(playerGui:GetChildren()) do
			if g:IsA("LayerCollector") then
				self.SavedGui[g] = g.Enabled
				g.Enabled = false
			end
		end
		self.SavedCore = {}
		for _, t in ipairs({Enum.CoreGuiType.Health, Enum.CoreGuiType.Backpack, Enum.CoreGuiType.Chat, Enum.CoreGuiType.PlayerList, Enum.CoreGuiType.EmotesMenu}) do
			pcall(function()
				self.SavedCore[t] = StarterGui:GetCoreGuiEnabled(t)
				StarterGui:SetCoreGuiEnabled(t, false)
			end)
		end
		log(L("ПАНИКА: всё скрыто", "PANIC: everything hidden"))
	else
		for g, was in pairs(self.SavedGui or {}) do
			pcall(function()
				if g and g.Parent then
					g.Enabled = was
				end
			end)
		end
		self.SavedGui = {}
		for t, was in pairs(self.SavedCore or {}) do
			pcall(function()
				StarterGui:SetCoreGuiEnabled(t, was)
			end)
		end
		self.SavedCore = {}
		log(L("Паника снята", "Panic released"))
	end
end

local KillcamMod = registerPort({
	name = "Killcam",
	ru = "Килл-кам",
	settings = {
		{type = "slider", name = "LookTime", ru = "Время (сек)", min = 0.5, max = 4, step = 0.1, default = 1.5},
	},
	onStart = function(env)
		function env:HookChar(char)
			local hum = char and char:WaitForChild("Humanoid", 5)
			if not hum then
				return
			end
			self:Connect(hum.Died:Connect(function()
				if self.Mod.enabled then
					task.spawn(function()
						self:OnDied(char)
					end)
				end
			end))
		end
		function env:OnDied(char)
			local cam = workspace.CurrentCamera
			if not cam then
				return
			end
			local myRoot = char and char:FindFirstChild("HumanoidRootPart")
			if not myRoot then
				return
			end
			local best, bestD = nil, math.huge
			for _, pl in ipairs(Players:GetPlayers()) do
				if pl ~= player then
					local c = pl.Character
					local h = c and c:FindFirstChildOfClass("Humanoid")
					local r = c and c:FindFirstChild("HumanoidRootPart")
					if h and r and h.Health > 0 then
						local d = (r.Position - myRoot.Position).Magnitude
						if d < bestD and d < 250 then
							best, bestD = pl, d
						end
					end
				end
			end
			if not best then
				log(L("Килл-кам: убийца не определён", "Killcam: killer not found"))
				return
			end
			local kRoot = best.Character and best.Character:FindFirstChild("HumanoidRootPart")
			if not kRoot then
				return
			end
			log(L("Килл-кам: смотрим на ", "Killcam: watching ") .. best.Name)
			if self.CamConn then
				pcall(function() self.CamConn:Disconnect() end)
				self.CamConn = nil
			end
			self._restoreType = cam.CameraType
			self._restoreSubject = cam.CameraSubject
			local dur = tonumber(self.State.LookTime) or 1.5
			cam.CameraType = Enum.CameraType.Scriptable
			local t0 = tick()
			self.CamConn = RunService.RenderStepped:Connect(function()
				local t = tick() - t0
				if t >= dur or not kRoot.Parent then
					if self.CamConn then
						self.CamConn:Disconnect()
						self.CamConn = nil
					end
					pcall(function()
						cam.CameraType = self._restoreType or Enum.CameraType.Custom
						cam.CameraSubject = self._restoreSubject
							or (player.Character and player.Character:FindFirstChildOfClass("Humanoid"))
					end)
					return
				end
				local myHead = char and char:FindFirstChild("Head")
				local from = myHead and myHead.Position or myRoot.Position
				local to = kRoot.Position + Vector3.new(0, 1.5, 0)
				local dir = to - from
				dir = dir.Magnitude > 0.01 and dir.Unit or Vector3.new(0, 0, -1)
				local back = 8 - 3 * math.min(1, t / dur)
				local pos = from - dir * back + Vector3.new(0, 2.5, 0)
				pcall(function()
					cam.CFrame = CFrame.lookAt(pos, to)
				end)
			end)
		end
		env:HookChar(player.Character)
		env:Connect(player.CharacterAdded:Connect(function(char)
			env:HookChar(char)
		end))
	end,
	onStop = function(env)
		if env.CamConn then
			pcall(function() env.CamConn:Disconnect() end)
			env.CamConn = nil
		end
		pcall(function()
			local cam = workspace.CurrentCamera
			if cam then
				cam.CameraType = Enum.CameraType.Custom
				cam.CameraSubject = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
			end
		end)

local CustomSoundsMod = registerPort({
	name = "Custom Sounds",
	ru = "Свои звуки",
	settings = {},
	onStart = function(env)
		-- подмена по категориям: вписай свои id в таблицу, чтобы включить
		env.MY_SOUNDS = {
			-- button = "rbxassetid://...",
			-- notify = "rbxassetid://...",
			-- menu = "rbxassetid://...",
			-- spawn = "rbxassetid://...",
			-- important = "rbxassetid://...",
		}
		function env:PickCategory(name)
			if not name then
				return nil
			end
			local n = name:lower()
			if n:find("important") or n:find("critical") or n:find("alert") or n:find("danger") or n:find("warning") or n:find("alarm") then
				return "important"
			end
			if n:find("spawn") or n:find("drop") or n:find("pickup") or n:find("collect") or n:find("buy") or n:find("reward") or n:find("unlock") then
				return "spawn"
			end
			if n:find("menu") then
				return "menu"
			end
			if n:find("notify") or n:find("notif") or n:find("info") or n:find("popup") or n:find("message") then
				return "notify"
			end
			if n:find("button") or n:find("btn") or n:find("click") or n:find("hover") or n:find("tap") or n:find("press") then
				return "button"
			end
			return nil
		end
		function env:Remap(snd)
			if not snd or not snd:IsA("Sound") then
				return
			end
			if snd:GetAttribute("FosforSFX") then
				return
			end
			local cat = self:PickCategory(snd.Name)
			if not cat then
				return
			end
			local newId = self.MY_SOUNDS[cat]
			if newId and snd.SoundId ~= newId then
				snd.SoundId = newId
			end
		end
		pcall(function()
			for _, d in ipairs(game:GetDescendants()) do
				if d:IsA("Sound") then
					env:Remap(d)
				end
			end
		end)
		env:Connect(workspace.DescendantAdded:Connect(function(d)
			if d:IsA("Sound") then
				env:Remap(d)
			end
		end))
		pcall(function()
			local ss = game:GetService("SoundService")
			env:Connect(ss.DescendantAdded:Connect(function(d)
				if d:IsA("Sound") then
					env:Remap(d)
				end
			end))
		end)
		log(L("Свои звуки: подмена по категориям активна (id в коде)", "Custom sounds: category remap active"))
	end,
	onStop = function(env)
		log(L("Свои звуки выключены", "Custom sounds off"))
	end,
})

local function trySetCap(value)
	local success = false
	pcall(function()
		if typeof(setfpscap) == "function" then
			setfpscap(value)
			success = true
			return
		end
		local env = genv()
		if env and typeof(env.setfpscap) == "function" then
			env.setfpscap(value)
			success = true
			return
		end
		if _G and typeof(_G.setfpscap) == "function" then
			_G.setfpscap(value)
			success = true
		end
	end)
	return success
end

local function buildSlider(parent, def, mod, order, ctx)
	local row = rowFrame(parent, 42, order)
	local lbl = text(row, settingLabel(def), 12, {Size = UDim2.new(0.5, 0, 0, 18), Position = UDim2.fromOffset(10, 4)})
	bindText(lbl, def.ru or def.name, def.name)
	local val = text(row, "", 12, {Size = UDim2.new(0.4, 0, 0, 18), Position = UDim2.new(0.6, -10, 0, 4), TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = ACCENT})
	local track = new("Frame", {Size = UDim2.new(1, -20, 0, 4), Position = UDim2.new(0, 10, 1, -13), BackgroundColor3 = WHITE, BackgroundTransparency = 0.8, BorderSizePixel = 0}, row)
	corner(track, 2)
	local fill = new("Frame", {Size = UDim2.fromScale(0, 1), BackgroundColor3 = WHITE, BorderSizePixel = 0}, track)
	corner(fill, 2)
	accentGradient(fill)
	local knob = new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(13, 13), Position = UDim2.fromScale(0, 0.5), BackgroundColor3 = WHITE, BorderSizePixel = 0}, track)
	corner(knob, 10)

	local function render(v)
		local a = (v - def.min) / (def.max - def.min)
		fill.Size = UDim2.fromScale(a, 1)
		knob.Position = UDim2.fromScale(a, 0.5)
		val.Text = (v % 1 == 0) and tostring(v) or string.format("%.2f", v)
	end
	render(mod.v[def.name])
	onAccent(function()
		fill.BackgroundColor3 = WHITE
		val.TextColor3 = ACCENT
	end)

	local function fromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		local v = def.min + a * (def.max - def.min)
		if def.step then
			v = math.floor(v / def.step + 0.5) * def.step
		end
		v = math.clamp(v, def.min, def.max)
		render(v)
		if mod.v[def.name] ~= v then
			mod.changed(def.name, v)
		end
	end

	local hit = hitButton(row)
	hit.InputBegan:Connect(function(i)
		if isPress(i) then
			fromX(i.Position.X)
			startDrag(function(pos)
				fromX(pos.X)
			end, nil, ctx.scroller)
		end
	end)
	return 42
end

local function buildStepped(parent, def, mod, order, ctx)
	local n = #def.options
	local row = rowFrame(parent, 62, order)
	local lbl = text(row, settingLabel(def), 12, {Size = UDim2.new(0.5, 0, 0, 18), Position = UDim2.fromOffset(10, 4)})
	bindText(lbl, def.ru or def.name, def.name)
	local val = text(row, "", 12, {Size = UDim2.new(0.4, 0, 0, 18), Position = UDim2.new(0.6, -10, 0, 4), TextXAlignment = Enum.TextXAlignment.Right, TextColor3 = ACCENT})
	local track = new("Frame", {Size = UDim2.new(1, -44, 0, 4), Position = UDim2.new(0, 22, 0, 31), BackgroundColor3 = WHITE, BackgroundTransparency = 0.8, BorderSizePixel = 0}, row)
	corner(track, 2)
	local fill = new("Frame", {Size = UDim2.fromScale(0, 1), BackgroundColor3 = ACCENT, BorderSizePixel = 0}, track)
	corner(fill, 2)

	local labels = {}
	for i = 1, n do
		local f = (i - 1) / (n - 1)
		local tickF = new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(8, 8), Position = UDim2.fromScale(f, 0.5), BackgroundColor3 = WHITE, BackgroundTransparency = 0.55, BorderSizePixel = 0}, track)
		corner(tickF, 8)
		local optRu = def.optionsRu and def.optionsRu[i] or def.options[i]
		labels[i] = text(row, L(optRu, def.options[i]), 11, {
			AnchorPoint = Vector2.new(0.5, 0),
			Size = UDim2.fromOffset(44, 14),
			Position = UDim2.new(f, 0, 0, 12),
			TextXAlignment = Enum.TextXAlignment.Center,
		})
		bindText(labels[i], optRu, def.options[i])
		labels[i].Parent = track
	end
	local knob = new("Frame", {AnchorPoint = Vector2.new(0.5, 0.5), Size = UDim2.fromOffset(15, 15), Position = UDim2.fromScale(0, 0.5), BackgroundColor3 = WHITE, BorderSizePixel = 0, ZIndex = 3}, track)
	corner(knob, 10)

	local function markIdx(idx)
		val.Text = L((def.optionsRu and def.optionsRu[idx]) or def.options[idx], def.options[idx])
		for i = 1, n do
			tw(labels[i], 0.2, {TextColor3 = i == idx and ACCENT or Color3.fromRGB(200, 206, 208)})
		end
	end
	local function place(frac, animate)
		if animate then
			tw(knob, 0.25, {Position = UDim2.fromScale(frac, 0.5)})
			tw(fill, 0.25, {Size = UDim2.fromScale(frac, 1)})
		else
			knob.Position = UDim2.fromScale(frac, 0.5)
			fill.Size = UDim2.fromScale(frac, 1)
		end
	end
	local cur = mod.v[def.name]
	place((cur - 1) / (n - 1), false)
	markIdx(cur)
	onAccent(function()
		fill.BackgroundColor3 = ACCENT
		markIdx(mod.v[def.name])
	end)

	local function fromX(x)
		local a = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
		place(a, false)
		local idx = math.floor(a * (n - 1) + 0.5) + 1
		if idx ~= mod.v[def.name] then
			markIdx(idx)
			mod.changed(def.name, idx)
		end
	end

	local hit = hitButton(row)
	hit.InputBegan:Connect(function(i)
		if isPress(i) then
			fromX(i.Position.X)
			startDrag(function(pos)
				fromX(pos.X)
			end, function()
				place((mod.v[def.name] - 1) / (n - 1), true)
			end, ctx.scroller)
		end
	end)
	return 62
end

local function buildModule(body, mod, order, ctx)
	local holderM = new("Frame", {Name = mod.name, Size = UDim2.new(1, 0, 0, 44), BackgroundTransparency = 1, ClipsDescendants = true, LayoutOrder = order}, body)
	mod.holder = holderM

	local row = new("Frame", {Position = UDim2.fromOffset(3, 3), Size = UDim2.new(1, -6, 0, 34), BackgroundColor3 = WHITE, BackgroundTransparency = 0.9, BorderSizePixel = 0}, holderM)
	corner(row, 10)
	local rowGrad = new("UIGradient", {Rotation = 0, Color = ColorSequence.new({ColorSequenceKeypoint.new(0, Color3.fromRGB(18, 20, 24)), ColorSequenceKeypoint.new(1, WHITE)})}, row)
	local rowScale = new("UIScale", {}, row)

	local nameBtn = new("TextButton", {
		Size = UDim2.new(1, -48, 1, 0),
		BackgroundTransparency = 1,
		Text = mod.title(),
		TextSize = 13,
		FontFace = FONT_REG,
		TextColor3 = Color3.fromRGB(215, 221, 224),
		TextXAlignment = Enum.TextXAlignment.Left,
		AutoButtonColor = false,
	}, row)
	bindText(nameBtn, mod.ru or mod.name, mod.name)
	new("UIPadding", {PaddingLeft = UDim.new(0, 12)}, nameBtn)

	local setBtn = new("TextButton", {Size = UDim2.new(0, 48, 1, 0), Position = UDim2.new(1, -48, 0, 0), BackgroundTransparency = 1, Text = "", AutoButtonColor = false, ZIndex = 3}, row)
	local bars = {}
	for i = 0, 2 do
		bars[i] = new("Frame", {
			AnchorPoint = Vector2.new(0.5, 0.5),
			Size = UDim2.fromOffset(14, 2),
			Position = UDim2.new(0.5, 0, 0.5, (i - 1) * 4),
			BackgroundColor3 = Color3.fromRGB(205, 211, 214),
			BorderSizePixel = 0,
			ZIndex = 4,
		}, setBtn)
		corner(bars[i], 2)
	end
	setBtn.MouseEnter:Connect(function()
		if not open then
			for _, b in pairs(bars) do
				tw(b, 0.15, {BackgroundColor3 = ACCENT})
			end
		end
	end)
	setBtn.MouseLeave:Connect(function()
		if not open then
			for _, b in pairs(bars) do
				tw(b, 0.15, {BackgroundColor3 = Color3.fromRGB(205, 211, 214)})
			end
		end
	end)

	-- настройки
	local sFrame = new("Frame", {Position = UDim2.fromOffset(3, 46), BackgroundTransparency = 1, Size = UDim2.new(1, -6, 0, 10), Visible = false}, holderM)
	new("UIListLayout", {Padding = UDim.new(0, 4), SortOrder = Enum.SortOrder.LayoutOrder}, sFrame)
	local total, count = 0, 0
	local function add(h)
		total += h
		count += 1
	end
	for i, def in ipairs(mod.settings) do
		if def.type == "slider" then
			add(buildSlider(sFrame, def, mod, i, ctx))
		elseif def.type == "stepped" then
			add(buildStepped(sFrame, def, mod, i, ctx))
		elseif def.type == "toggle" then
			add(buildToggle(sFrame, def, mod, i))
		elseif def.type == "list" then
			add(buildList(sFrame, def, mod, i))
		elseif def.type == "color" then
			add(buildColor(sFrame, def, mod, i, ctx))
		elseif def.type == "input" then
			add(buildInput(sFrame, def, mod, i))
		elseif def.type == "gallery" then
			add(buildGallery(sFrame, def, mod, i))
		elseif def.type == "button" then
			add(buildButton(sFrame, def, mod, i))
		elseif def.type == "menubind" then
			add(buildBind(sFrame, menuProxy, i, settingLabel(def)))
		end
	end
	if not mod.noBind then
		add(buildBind(sFrame, mod, 100))
	end
	total += (count - 1) * 4
	sFrame.Size = UDim2.new(1, -6, 0, total)
	local openH = 46 + total + 6

	local sRows = {}
	for _, c in ipairs(sFrame:GetChildren()) do
		if c:IsA("GuiObject") then
			table.insert(sRows, c)
		end
	end

	local hover, open = false, false
	function mod.setVisual(on)
		tw(row, 0.28, {BackgroundColor3 = WHITE, BackgroundTransparency = on and 0.25 or (hover and 0.84 or 0.9)})
		tw(nameBtn, 0.28, {TextColor3 = WHITE})
		pcall(function()
			rowGrad.Enabled = on
		end)
	end

	local function toggleOpen()
		open = not open
		sFrame.Visible = open
		if open then
			for _, r in ipairs(sRows) do
				r.Visible = false
			end
			task.delay(0.2, function()
				for i, r in ipairs(sRows) do
					task.delay((i - 1) * 0.025, function()
						if open and r.Parent then
							r.Visible = true
						end
					end)
				end
			end)
		else
			for _, r in ipairs(sRows) do
				r.Visible = false
			end
		end
		tw(holderM, 0.18, {Size = UDim2.new(1, 0, 0, open and openH or 44)})
		for _, b in pairs(bars) do
			tw(b, 0.25, {BackgroundColor3 = open and ACCENT or Color3.fromRGB(205, 211, 214)})
		end
	end

	function mod.getH()
		return open and openH or 44
	end

	nameBtn.MouseButton1Click:Connect(function()
		if mod.noToggle then
			toggleOpen()
		else
			local turningOn = not mod.enabled
			setEnabled(mod, turningOn)
			if turningOn ~= open then
				toggleOpen()
			end
		end
	end)
	nameBtn.MouseButton2Click:Connect(toggleOpen)
	setBtn.MouseButton1Click:Connect(toggleOpen)
	onAccent(function()
		if mod.enabled and mod.setVisual then
			mod.setVisual(true)
		end
		for _, b in pairs(bars) do
			b.BackgroundColor3 = open and ACCENT or Color3.fromRGB(205, 211, 214)
		end
	end)

	nameBtn.MouseButton1Down:Connect(function()
		tw(rowScale, 0.12, {Scale = 0.97})
	end)
	local function release()
		tw(rowScale, 0.2, {Scale = 1}, Enum.EasingStyle.Back)
	end
	nameBtn.MouseButton1Up:Connect(release)
	nameBtn.MouseLeave:Connect(function()
		hover = false
		release()
		if not mod.enabled then
			tw(row, 0.2, {BackgroundTransparency = 0.9})
		end
	end)
	nameBtn.MouseEnter:Connect(function()
		hover = true
		if not mod.enabled then
			tw(row, 0.2, {BackgroundTransparency = 0.84})
		end
	end)
end

-- адаптация к экрану: смотрим постоянные UI игры и отступаем от них сверху
local topSafe = 50
local function scanForeignUI()
	local blocked = 30
	pcall(function()
		for _, g in ipairs(playerGui:GetChildren()) do
			if g:IsA("LayerCollector") and g.Enabled and g ~= gui and g ~= dimGui and g ~= hud
				and g.Name:sub(1, 13) ~= "FosforVisuals" then
				local n = 0
				for _, obj in ipairs(g:GetChildren()) do
					n += 1
					if n > 40 then
						break
					end
					if obj:IsA("GuiObject") and obj.Visible then
						local ap, as = obj.AbsolutePosition, obj.AbsoluteSize
						if ap.Y < 320 and as.Y > 8 then
							local bottom = ap.Y + as.Y
							if bottom > blocked then
								blocked = bottom
							end
						end
					end
				end
			end
		end
	end)
	topSafe = math.clamp(blocked / math.max(scale, 0.1) + 14, 44, 90)
end

	local function applySearch(q)
		q = (q or ""):lower():match("^%s*(.-)%s*$")
		for _, p in ipairs(panels) do
			local tabHit = q ~= "" and p.name:lower():find(q, 1, true) ~= nil
			local any = false
			for _, m in ipairs(p.mods) do
				local match = q == "" or tabHit or m.name:lower():find(q, 1, true) ~= nil or (m.ru and m.ru:lower():find(q, 1, true) ~= nil)
				if match then
					any = true
				end
				if match == (m.hidden == true) then
					m.hidden = not match
					if match then
						m.holder.Visible = true
						tw(m.holder, 0.38, {Size = UDim2.new(1, 0, 0, m.getH())})
					else
						local t = tw(m.holder, 0.3, {Size = UDim2.new(1, 0, 0, 0)})
						t.Completed:Connect(function(state)
							if state == Enum.PlaybackState.Completed and m.hidden then
								m.holder.Visible = false
							end
						end)
					end
				end
			end
			p.searchEmpty = q ~= "" and not any
		end
	end

	-- мягкие столкновения: панели не проходят друг сквозь друга
	local function resolvePanels()
		for _ = 1, 6 do
			local pushed = false
			for _, a in ipairs(panels) do
				for _, b in ipairs(panels) do
					if a ~= b and a.appear < 0.6 and b.appear < 0.6 then -- appear=0 ВИДИМА, 1 скрыта
						if b == dragPanel then
							continue -- тащимую не толкаем: её вытолкнет пара (dragP, X)
						end
						local ax1, ax2 = a.tx - 95, a.tx + 95
						local ay1, ay2 = a.ty, a.ty + a.frame.Size.Y.Offset
						local bx1, bx2 = b.tx - 95, b.tx + 95
						local by1, by2 = b.ty, b.ty + b.frame.Size.Y.Offset
						local pushX = math.min(ax2, bx2) - math.max(ax1, bx1)
						local pushY = math.min(ay2, by2) - math.max(ay1, by1)
						if pushX > 0 and pushY > 0 then
							local victim, blocker = b, a
							if b == dragPanel and a ~= dragPanel then
								victim, blocker = a, b -- тащат одну — другая уступает
							end
							local vCx = victim.tx
							local vCy = victim.ty + victim.frame.Size.Y.Offset / 2
							if pushY <= pushX then
								if vCy >= blocker.ty + blocker.frame.Size.Y.Offset / 2 then
									victim.ty = victim.ty + pushY + 4
								else
									victim.ty = victim.ty - pushY - 4
								end
							else
								if vCx >= blocker.tx then
									victim.tx = victim.tx + pushX + 4
								else
									victim.tx = victim.tx - pushX - 4
								end
							end
							victim.tx = clampX(victim.tx)
							victim.ty = math.clamp(victim.ty, 6, holder.AbsoluteSize.Y / math.max(scale, 0.1) - 70)
							pushed = true
						end
					end
				end
			end
			if not pushed then
				break
			end
		end
	end

	RunService.RenderStepped:Connect(function(dt)
		resolvePanels()
		local appearK = 1 - math.exp(-dt * 9)
		local holderH = holder.AbsoluteSize.Y / scale
		local allHidden = true
		for _, p in ipairs(panels) do
			p.k = (p.k or 16) + (16 - (p.k or 16)) * math.min(1, dt * 5)
			local follow = 1 - math.exp(-dt * (p.k or 16))
			if p.wait > 0 then
				p.wait -= dt
			else
				p.appear += (p.target - p.appear) * appearK
			end
			p.cx += (p.tx - p.cx) * follow
			p.cy += (p.ty - p.cy) * follow

			local content, visN = 0, 0
			for _, m in ipairs(p.mods) do
				if m.holder.Visible then
					content += m.holder.Size.Y.Offset
					visN += 1
				end
			end
			content += math.max(visN - 1, 0) * 2
			if visN == 0 then
				content = 30
			end
			local avail = math.max(90, holderH - p.cy - 40 - 66)
			local bodyH = math.min(content, avail)
			p.col += (((p.collapsed or p.searchEmpty) and 1 or 0) - p.col) * (1 - math.exp(-dt * 12))
			local openK = 1 - p.col
			p.body.Size = UDim2.new(1, -12, 0, bodyH * openK)
			p.body.CanvasSize = UDim2.fromOffset(0, content)
			p.body.Visible = openK > 0.03
			p.frame.Size = UDim2.fromOffset(190, 40 + (6 + bodyH + 10) * openK)

			p.dim += ((p.searchEmpty and 0.62 or 0) - p.dim) * (1 - math.exp(-dt * 10))
			p.frame.GroupTransparency = math.clamp(math.max(p.appear, p.dim), 0, 1)
			p.frame.Position = UDim2.new(0.5, p.cx, 0, p.cy + p.appear * 26)
			if p.appear < 0.98 then
				allHidden = false
			end
		end
		local sb = holder:FindFirstChild("Search")
		if sb then
			local ap = panels[1] and panels[1].appear or 1
			sb.GroupTransparency = math.clamp(ap, 0, 1)
			sb.Position = UDim2.new(0.5, 0, 1, -12 + ap * 20)
		end
		if not menuOpen and allHidden then
			holder.Visible = false
		end
	end)
end

UIS.InputBegan:Connect(function(input, gpe)
	if listening then
		if input.UserInputType == Enum.UserInputType.Keyboard then
			local l = listening
			listening = nil
			if input.KeyCode == Enum.KeyCode.Escape or input.KeyCode == Enum.KeyCode.Backspace then
				l.mod.key = nil
			else
				l.mod.key = input.KeyCode
			end
			l.update()
			refreshKeybinds()
		end
		return
	end
	if input.UserInputType ~= Enum.UserInputType.Keyboard then
		return
	end
	if input.KeyCode == MENU_KEY and not UIS:GetFocusedTextBox() then
		setMenu(not menuOpen)
		return
	end
	if UIS:GetFocusedTextBox() then
		return
	end
	for _, m in ipairs(modules) do
		if m.key and m.key == input.KeyCode then
			if m.PanicAction then
				m:PanicAction()
			else
				setEnabled(m, not m.enabled)
			end
		end
	end
end)

-- СТАРТ: интро -> (диалог, если не FTAP) -> сборка меню -> открытие
--------------------------------------------------------------------
task.spawn(function()
	pcall(playIntro)
	if not isFTAP then
		local ans = showFtapDialog()
		if ans == "universal" then
			includeFTAP = false
			log(L("Универсальный режим: вкладка FTAP скрыта", "Universal mode: FTAP tab hidden"))
		else
			includeFTAP = true
			log(L("Полная загрузка: вкладка FTAP включена", "Full load: FTAP tab enabled"))
		end
	end
	if includeFTAP then
		FTAP_MODS, FTAP_BYNAME = defineFTAP()
	end
	pcall(buildMenu)
	refreshKeybinds()
	task.delay(0.5, function()
		setMenu(true)
	end)
end)
