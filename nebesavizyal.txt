local function CreateTab(name, icon)

    TabBtn.BackgroundColor3 = Theme.Sidebar
    TabBtn.Text = "  " .. icon .. "  " .. name

    TabBtn.TextXAlignment = Enum.TextXAlignment.Left
    TabBtn.AutoButtonColor = false
    TabBtn.Parent = Sidebar
    Instance.new("UICorner", TabBtn).CornerRadius = UDim.new(0, 6)

    local ScrollFrame = Instance.new("ScrollingFrame")
    ScrollFrame.Size = UDim2.new(1, 0, 1, 0)
    ScrollFrame.BackgroundTransparency = 1
    ScrollFrame.BorderSizePixel = 0
    ScrollFrame.ScrollBarThickness = 3
    ScrollFrame.ScrollBarImageColor3 = Color3.fromRGB(80, 80, 90)
    ScrollFrame.Visible = false
    ScrollFrame.Parent = ContentArea

    local Layout = Instance.new("UIListLayout")
    Layout.Padding = UDim.new(0, 8)
    Layout.Parent = ScrollFrame

    Layout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        ScrollFrame.CanvasSize = UDim2.new(0, 0, 0, Layout.AbsoluteContentSize.Y + 20)
    end)

    Tabs[name] = { Btn = TabBtn, Frame = ScrollFrame }

    TabBtn.MouseEnter:Connect(function()
        if ActiveTab ~= name then TabBtn.BackgroundColor3 = Theme.Element end
    end)
    TabBtn.MouseLeave:Connect(function()
        if ActiveTab ~= name then TabBtn.BackgroundColor3 = Theme.Sidebar end
    end)

    TabBtn.MouseButton1Click:Connect(function()
        for tName, tData in pairs(Tabs) do
            tData.Btn.BackgroundColor3 = Theme.Sidebar
            tData.Btn.TextColor3 = Theme.SubText
            tData.Frame.Visible = false
        end
        TabBtn.BackgroundColor3 = Theme.Element
        TabBtn.TextColor3 = Theme.Text
        ScrollFrame.Visible = true
        ActiveTab = name
    end)

    return ScrollFrame, TabBtn
end

local function AddSlider(parent, text, min, max, default, callback)
    local Frame = Instance.new("Frame")
    Frame.Size = UDim2.new(1, -10, 0, 46)
    Frame.BackgroundColor3 = Theme.Element
    Frame.Parent = parent
    Instance.new("UICorner", Frame).CornerRadius = UDim.new(0, 6)
    local Label = Instance.new("TextLabel")
    Label.Size = UDim2.new(1, -20, 0, 20)
    Label.Position = UDim2.new(0, 12, 0, 4)
    Label.BackgroundTransparency = 1
    Label.Text = text .. ": " .. tostring(default)
    Label.TextColor3 = Theme.Text
    Label.Font = Enum.Font.Gotham
    Label.TextSize = 13
    Label.TextXAlignment = Enum.TextXAlignment.Left
    Label.Parent = Frame
    local Bar = Instance.new("Frame")
    Bar.Size = UDim2.new(1, -24, 0, 4)
    Bar.Position = UDim2.new(0, 12, 0, 30)
    Bar.BackgroundColor3 = Color3.fromRGB(50, 50, 60)
    Bar.Parent = Frame
    Instance.new("UICorner", Bar).CornerRadius = UDim.new(1, 0)
    local Fill = Instance.new("Frame")
    Fill.Size = UDim2.new((default - min) / (max - min), 0, 1, 0)
    Fill.BackgroundColor3 = Theme.Accent
    Fill.Parent = Bar
    Instance.new("UICorner", Fill).CornerRadius = UDim.new(1, 0)
    local Knob = Instance.new("Frame")
    Knob.Size = UDim2.new(0, 12, 0, 12)
    Knob.Position = UDim2.new((default - min) / (max - min), -6, 0.5, -6)
    Knob.BackgroundColor3 = Theme.Text
    Knob.Parent = Bar
    Instance.new("UICorner", Knob).CornerRadius = UDim.new(1, 0)
    local Dragging = false
    local function Update(input)
        if Bar.AbsoluteSize.X <= 0 then return end
        local relX = math.clamp(input.Position.X - Bar.AbsolutePosition.X, 0, Bar.AbsoluteSize.X)
        local percent = relX / Bar.AbsoluteSize.X
        local value = math.floor(min + (max - min) * percent)
        Fill.Size = UDim2.new(percent, 0, 1, 0)
        Knob.Position = UDim2.new(percent, -6, 0.5, -6)
        Label.Text = text .. ": " .. tostring(value)
        pcall(function() callback(value) end)
    end
    Bar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = true
            Update(input)
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if Dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            Update(input)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = false
        end
    end)
end

local function MakeDraggable(DragPart, Target)
    local Dragging, DragStart, StartPos = false, nil, nil
    DragPart.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = true
            DragStart = input.Position
            StartPos = Target.Position
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if Dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - DragStart
            Target.Position = UDim2.new(StartPos.X.Scale, StartPos.X.Offset + delta.X, StartPos.Y.Scale, StartPos.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            Dragging = false
        end
    end)
end

local function SetSky(skyName)
    local id = SkyboxData[skyName]
    local sky = Lighting:FindFirstChild("Nebesny_Sky")
    if skyName == "Roblox Default" then
        if sky then sky:Destroy() end
        return
    end
    if not sky then
        sky = Instance.new("Sky")
        sky.Name = "Nebesny_Sky"
        sky.Parent = Lighting
    end
    if type(id) == "table" then
        sky.SkyboxBk = "rbxassetid://" .. id[1]
        sky.SkyboxDn = "rbxassetid://" .. id[2]
        sky.SkyboxFt = "rbxassetid://" .. id[3]
        sky.SkyboxLf = "rbxassetid://" .. id[4]
        sky.SkyboxRt = "rbxassetid://" .. id[5]
        sky.SkyboxUp = "rbxassetid://" .. id[6]
    else
        local asset = "rbxassetid://" .. tostring(id)
        sky.SkyboxBk = asset; sky.SkyboxDn = asset; sky.SkyboxFt = asset
        sky.SkyboxLf = asset; sky.SkyboxRt = asset; sky.SkyboxUp = asset
    end
end

local Shaders = {}
local function ToggleShader(name, className, props, state)
    if not Shaders[name] then
        local obj = Instance.new(className)
        obj.Name = "Nebesny_" .. name
        for k, v in pairs(props) do pcall(function() obj[k] = v end) end
        obj.Parent = Lighting
        Shaders[name] = obj
    end
    Shaders[name].Enabled = state
    if name == "Bloom" and state then Lighting.Brightness = 3 end
end

local function ToggleOptimize(state)
    if state then
        Lighting.Brightness = 3
        Lighting.Ambient = Color3.fromRGB(255, 255, 255)
        Lighting.OutdoorAmbient = Color3.fromRGB(255, 255, 255)
        Lighting.FogEnd = 1e9
        Lighting.GlobalShadows = false
        for _, v in ipairs(Lighting:GetChildren()) do
            if v:IsA("PostEffect") then v.Enabled = false end
        end
    else
        Lighting.Brightness = 2
        Lighting.Ambient = Color3.fromRGB(70, 70, 70)
        Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
        Lighting.FogEnd = 100000
        Lighting.GlobalShadows = true
    end
end

local JumpCirclePart = nil
local TraceLine = nil

local function ToggleJumpCircle(state)
    if state then
        if not JumpCirclePart then
            JumpCirclePart = Instance.new("Part")
            JumpCirclePart.Size = Vector3.new(8, 0.5, 8)
            JumpCirclePart.Shape = Enum.PartType.Cylinder
            JumpCirclePart.Material = Enum.Material.Neon
            JumpCirclePart.Color = Color3.fromRGB(0, 255, 255)
            JumpCirclePart.CanCollide = false
            JumpCirclePart.Anchored = true
            JumpCirclePart.Transparency = 0.4
            JumpCirclePart.Parent = Workspace
        end
        JumpCirclePart.Transparency = 0.4
    else
        if JumpCirclePart then JumpCirclePart.Transparency = 1 end
    end
end

RunService.Heartbeat:Connect(function()
    if JumpCirclePart and JumpCirclePart.Transparency < 1 then
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if hrp then
            JumpCirclePart.CFrame = CFrame.new(hrp.Position - Vector3.new(0, 2.8, 0)) * CFrame.Angles(0, 0, math.rad(90))
        end
    end
end)

local function ToggleTrace(state)
    if state then
        if not TraceLine then
            TraceLine = Instance.new("Part")
            TraceLine.Anchored = true
            TraceLine.CanCollide = false
            TraceLine.Material = Enum.Material.Neon
            TraceLine.Color = Color3.fromRGB(255, 0, 0)
            TraceLine.Parent = Workspace
        end
        TraceLine.Transparency = 0.5
    else
        if TraceLine then TraceLine.Transparency = 1 end
    end
end

RunService.Heartbeat:Connect(function()
    if TraceLine and TraceLine.Transparency < 1 then
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local closestPlayer, shortestDist = nil, math.huge
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local dist = (player.Character.HumanoidRootPart.Position - hrp.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    closestPlayer = player
                end
            end
        end
        if hrp and closestPlayer then
            local targetPos = closestPlayer.Character.HumanoidRootPart.Position
            local distance = (targetPos - hrp.Position).Magnitude
            TraceLine.Size = Vector3.new(0.2, 0.2, distance)
            TraceLine.CFrame = CFrame.lookAt(hrp.Position, targetPos) * CFrame.new(0, 0, -distance/2)
        else
            TraceLine.Transparency = 1
        end
    end
end)

FPSLbl.Size = UDim2.new(1, 0, 0.5, 0); FPSLbl.BackgroundTransparency = 1; FPSLbl.Text = "FPS: 60"; FPSLbl.TextColor3 = Theme.Accent; FPSLbl.Font = Enum.Font.GothamBold; FPSLbl.TextSize = 13; FPSLbl.Parent = StatsFrame

local PingLbl = Instance.new("TextLabel")
PingLbl.Size = UDim2.new(1, 0, 0.5, 0); PingLbl.Position = UDim2.new(0, 0, 0.5, 0); PingLbl.BackgroundTransparency = 1; PingLbl.Text = "Ping: 0 ms"; PingLbl.TextColor3 = Theme.Text; PingLbl.Font = Enum.Font.GothamBold; PingLbl.TextSize = 13; PingLbl.Parent = StatsFrame

MakeDraggable(StatsFrame, StatsFrame)

local Frames = 0
RunService.RenderStepped:Connect(function() Frames = Frames + 1 end)

task.spawn(function()
    while task.wait(1) do
        FPSLbl.Text = "FPS: " .. Frames
        FPSLbl.TextColor3 = Frames >= 50 and Theme.Accent or (Frames >= 30 and Color3.fromRGB(230, 200, 80) or Color3.fromRGB(230, 80, 80))
        Frames = 0
    end
end)

task.spawn(function()
    while task.wait(1) do
        pcall(function()
            local ping = 0
            local stat = game:GetService("Stats").Network.ServerStatsItem["Data Ping"]
            if stat then ping = stat:GetValue() end
            PingLbl.Text = "Ping: " .. math.floor(ping) .. " ms"
            PingLbl.TextColor3 = ping <= 80 and Theme.Accent or (ping <= 150 and Color3.fromRGB(230, 200, 80) or Color3.fromRGB(230, 80, 80))
        end)
    end
end)

print("[Nebesny Visual] Успешно загружено!")
